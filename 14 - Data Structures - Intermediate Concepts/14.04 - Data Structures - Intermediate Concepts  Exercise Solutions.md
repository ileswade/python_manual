### Question 1: 3×3 Matrix Operations
**Problem:** Create a 3×3 matrix (2D list) filled with numbers 1-9. Write a function that prints it in grid format and calculates the sum of each row.

**Solution Approach:**
- Create 2D list: list of lists where each inner list is a row
- Print with proper formatting for grid appearance
- Sum each row by iterating through the matrix

**Complete Solution:**
```python
def create_matrix():
    """Create 3x3 matrix with numbers 1-9."""
    matrix = [
        [1, 2, 3],
        [4, 5, 6],
        [7, 8, 9]
    ]
    return matrix


def print_matrix(matrix):
    """Print matrix in grid format."""
    print("\nMatrix Grid:")
    print("-" * 13)
    for row in matrix:
        print("|", end="")
        for num in row:
            print(f" {num} |", end="")
        print()
        print("-" * 13)


def calculate_row_sums(matrix):
    """Calculate and display sum of each row."""
    print("\nRow Sums:")
    for i, row in enumerate(matrix):
        row_sum = sum(row)
        print(f"Row {i+1}: {row} = {row_sum}")
    
    # Calculate total sum
    total = sum(sum(row) for row in matrix)
    print(f"\nTotal sum of all elements: {total}")


def calculate_column_sums(matrix):
    """Bonus: Calculate sum of each column."""
    print("\nColumn Sums:")
    num_cols = len(matrix[0])
    
    for col in range(num_cols):
        col_sum = sum(matrix[row][col] for row in range(len(matrix)))
        col_values = [matrix[row][col] for row in range(len(matrix))]
        print(f"Column {col+1}: {col_values} = {col_sum}")


# Test the functions

calculate_column_sums(matrix)

anti_diagonal = [matrix[i][2-i] for i in range(3)]
print(f"\nMain diagonal: {main_diagonal} = {sum(main_diagonal)}")
print(f"Anti-diagonal: {anti_diagonal} = {sum(anti_diagonal)}")
```

### Question 3: City Distance Finder

**Problem:** Create a list of tuples where each tuple contains a city name and its coordinates (lat, lon). Write a function that returns the city closest to a given coordinate.

**Solution Approach:**

- Store cities as list of (name, (lat, lon)) tuples
- Calculate distance from target to each city
- Track minimum distance and corresponding city

**Complete Solution:**

```python
import math

def haversine_distance(coord1, coord2):
    """Calculate distance between two GPS coordinates in km."""
    lat1, lon1 = coord1
    lat2, lon2 = coord2
    
    R = 6371  # Earth's radius in km
    
    # Convert to radians
    lat1, lon1, lat2, lon2 = map(math.radians, [lat1, lon1, lat2, lon2])
    
    # Haversine formula
    dlat = lat2 - lat1
    dlon = lon2 - lon1
    a = math.sin(dlat/2)**2 + math.cos(lat1) * math.cos(lat2) * math.sin(dlon/2)**2
    c = 2 * math.asin(math.sqrt(a))
    
    return R * c


def find_closest_city(target_coords, cities):
    """Find the city closest to target coordinates."""
    closest_city = None
    min_distance = float('inf')
    
    for city_name, city_coords in cities:
        distance = haversine_distance(target_coords, city_coords)
        
        if distance < min_distance:
            min_distance = distance
            closest_city = city_name
    
    return closest_city, min_distance


def display_all_distances(target_coords, cities, target_name="Target"):
    """Display distances from target to all cities."""
    print(f"\nDistances from {target_name} {target_coords}:")
    print("-" * 60)
    
    distances = []
    for city_name, city_coords in cities:
        distance = haversine_distance(target_coords, city_coords)
        distances.append((city_name, distance))
    
    # Sort by distance
    distances.sort(key=lambda x: x[1])
    
    for city_name, distance in distances:
        print(f"{city_name:20} : {distance:8.2f} km")
    
    return distances


def find_cities_within_radius(target_coords, cities, radius_km):
    """Find all cities within specified radius."""
    cities_in_range = []
    
    for city_name, city_coords in cities:
        distance = haversine_distance(target_coords, city_coords)
        if distance <= radius_km:
            cities_in_range.append((city_name, distance))
    
    return cities_in_range
```

**Problem:** Build a dictionary where keys are course names and values are lists of student dictionaries (each with name and grade). Calculate the average grade for each course.

**Solution Approach:**
- Structure: {course_name: [{name: ..., grade: ...}, ...]}
- Calculate average by summing grades and dividing by count
- Handle empty courses gracefully

**Complete Solution:**

```python
def calculate_course_averages(courses):
    """Calculate and display average grade for each course."""
    print("\n" + "="*60)
    print("COURSE GRADE AVERAGES")
    print("="*60)
    
    for course_name, students in courses.items():
        print(f"\n{course_name}:")

        if not students:

            continue
        
        # Display all students
        total_grade = 0

    """Find highest-scoring student in each course."""
    print("\n" + "="*60)
    print("TOP STUDENTS BY COURSE")
    print("="*60)
    
    for course_name, students in courses.items():
        if not students:
            print(f"\n{course_name}: No students enrolled")
            continue
        
        top_student = max(students, key=lambda s: s['grade'])
        print(f"\n{course_name}:")
        print(f"  {top_student['name']} - {top_student['grade']} points")


def find_student_gpa(courses, student_name):
    """Calculate GPA for a specific student across all courses."""
    student_grades = []
    enrolled_courses = []
    
    for course_name, students in courses.items():
        for student in students:
            if student['name'].lower() == student_name.lower():
                student_grades.append(student['grade'])
                enrolled_courses.append(course_name)
    
    if not student_grades:
        print(f"\n{student_name} is not enrolled in any courses")
        return
    
    gpa = sum(student_grades) / len(student_grades)
    
    print(f"\n{'='*60}")
    print(f"GPA REPORT FOR {student_name.upper()}")
    print('='*60)
    
    for course, grade in zip(enrolled_courses, student_grades):
        print(f"{course:25} : {grade}")
    
    print(f"{'-'*60}")
    print(f"Overall GPA: {gpa:.2f}")
    print(f"Courses taken: {len(enrolled_courses)}")


def get_course_statistics(courses):

            print(f"  Range: {min_grade} - {max_grade}")
            
            total_enrollments += len(students)
            for student in students:
                total_students.add(student['name'])
    
    print(f"\n{'='*60}")
    print(f"Total unique students: {len(total_students)}")
    print(f"Total enrollments: {total_enrollments}")
```

```python
import json

def convert_tuples_to_lists(data):
    """
    Recursively convert all tuples to lists for JSON compatibility.
    """
    if isinstance(data, dict):
        return {key: convert_tuples_to_lists(value) for key, value in data.items()}
    elif isinstance(data, list):
        return [convert_tuples_to_lists(item) for item in data]
    elif isinstance(data, tuple):
        return [convert_tuples_to_lists(item) for item in data]
    else:
        return data
```

```python
def merge_configs(base, override):
    """
    Recursively merge two configuration dictionaries.
    Override values take precedence over base values.
    """

            # Recursively merge nested dictionaries
            merged[key] = merge_configs(merged[key], value)
        else:
            # Override or add new key
            merged[key] = value
    
    return merged
```

```python
import math

def haversine_distance(coord1, coord2):
    """Calculate distance between two GPS coordinates in km."""
    lat1, lon1 = coord1
    lat2, lon2 = coord2
    
    R = 6371  # Earth's radius in km
    
    # Convert to radians
    lat1, lon1, lat2, lon2 = map(math.radians, [lat1, lon1, lat2, lon2])
    
    # Haversine formula
    dlat = lat2 - lat1
    dlon = lon2 - lon1
    a = math.sin(dlat/2)**2 + math.cos(lat1) * math.cos(lat2) * math.sin(dlon/2)**2
    c = 2 * math.asin(math.sqrt(a))
    
    return R * c


def calculate_distance_matrix(cities):
    """Calculate distances between all pairs of cities."""
    distances = {}
    
    city_list = list(cities.items())
    
    for i in range(len(city_list)):
        for j in range(i + 1, len(city_list)):
            city1_name, city1_coord = city_list[i]
            city2_name, city2_coord = city_list[j]
            
            dist = haversine_distance(city1_coord, city2_coord)
            pair = (city1_name, city2_name)
            distances[pair] = dist
    
    return distances
```

### Question 12: Cardinal Direction Point Calculator

**Problem:** Write a function that takes a starting coordinate and finds a point 5km north, south, east, and west. Return all four points as tuples.

**Solution Approach:**

- Calculate new coordinates based on cardinal directions
- Approximate: 1° latitude ≈ 111km, 1° longitude varies by latitude
- Use cosine for longitude adjustment based on latitude

**Complete Solution:**

```python
import math

def calculate_cardinal_points(start_coord, distance_km):
    """
    Calculate points at given distance in four cardinal directions.
    
    Parameters:
        start_coord: (lat, lon) tuple
        distance_km: Distance in kilometers
    
    Returns:
        Dictionary with north, south, east, west coordinates
    """
    lat, lon = start_coord
    
    # Degrees per km (approximate)
    lat_degrees_per_km = 1 / 111.0
    
    # Longitude degrees per km varies with latitude
    lon_degrees_per_km = 1 / (111.0 * math.cos(math.radians(lat)))
    
    # Calculate new coordinates
    points = {
        'north': (lat + (distance_km * lat_degrees_per_km), lon),
        'south': (lat - (distance_km * lat_degrees_per_km), lon),
        'east': (lat, lon + (distance_km * lon_degrees_per_km)),
        'west': (lat, lon - (distance_km * lon_degrees_per_km))
    }
    
    return points


def haversine_distance(coord1, coord2):
```

**Solution Approach:**

- Store tracking points as list of tuples
- Calculate distances between consecutive points
- Calculate time differences for speed
- Accumulate totals

**Complete Solution:**

```python
import math
from datetime import datetime, timedelta

def haversine_distance(coord1, coord2):
    """Calculate distance between coordinates in km."""
    lat1, lon1 = coord1
    lat2, lon2 = coord2
    
    R = 6371
    lat1, lon1, lat2, lon2 = map(math.radians, [lat1, lon1, lat2, lon2])
    
    dlat = lat2 - lat1
    dlon = lon2 - lon1
    a = math.sin(dlat/2)**2 + math.cos(lat1) * math.cos(lat2) * math.sin(dlon/2)**2
    c = 2 * math.asin(math.sqrt(a))
    
    return R * c


class GPSTracker:
    """GPS tracking system with distance and speed calculations."""
    
    def __init__(self):
        self.tracking_points = []
    
    def add_point(self, timestamp, coordinates):
        """Add a tracking point."""
        self.tracking_points.append((timestamp, coordinates))
    
    def calculate_total_distance(self):
        """Calculate total distance traveled."""
        if len(self.tracking_points) < 2:
            return 0
        
        total_distance = 0
        
        for i in range(1, len(self.tracking_points)):
            coord1 = self.tracking_points[i-1][1]
            coord2 = self.tracking_points[i][1]
            
            distance = haversine_distance(coord1, coord2)
            total_distance += distance
        
        return total_distance
    
    def calculate_average_speed(self):
        """Calculate average speed in km/h."""
        if len(self.tracking_points) < 2:
            return 0
        
        total_distance = self.calculate_total_distance()
        
        start_time = self.tracking_points[0][0]
        end_time = self.tracking_points[-1][0]
        
        time_diff = (end_time - start_time).total_seconds() / 3600  # hours
        
        if time_diff == 0:
            return 0
        
        return total_distance / time_diff
    
    def get_segment_speeds(self):
        """Calculate speed for each segment."""
        segments = []
        
        for i in range(1, len(self.tracking_points)):
            time1, coord1 = self.tracking_points[i-1]
            time2, coord2 = self.tracking_points[i]
            
            distance = haversine_distance(coord1, coord2)
            time_diff = (time2 - time1).total_seconds() / 3600  # hours
            
            speed = distance / time_diff if time_diff > 0 else 0
            
            segments.append({
                'from': coord1,
                'to': coord2,
                'distance': distance,
                'time': time_diff * 60,  # minutes
                'speed': speed
            })
        
        return segments
    
    def display_tracking_summary(self):

        avg_speed = self.calculate_average_speed()
        
        print(f"\nTotal Distance: {total_distance:.2f} km")
```

### Question 14: Boundary Detection System

**Problem:** Build a function that determines if a given coordinate is within a rectangular boundary defined by two corner coordinates (NW and SE corners).

```python
def is_within_boundary(point, nw_corner, se_corner):
    """
    Check if point is within rectangular boundary.
    
    Parameters:
        point: (lat, lon) to check
        nw_corner: (lat, lon) of northwest corner
        se_corner: (lat, lon) of southeast corner
    
    Returns:
        Boolean indicating if point is within boundary
    """
    point_lat, point_lon = point
    nw_lat, nw_lon = nw_corner
    se_lat, se_lon = se_corner
    
    # Northwest has higher latitude, lower longitude
    # Southeast has lower latitude, higher longitude
    max_lat = nw_lat
    min_lat = se_lat
    min_lon = nw_lon
    max_lon = se_lon
    
    lat_in_range = min_lat <= point_lat <= max_lat
    lon_in_range = min_lon <= point_lon <= max_lon
    
    return lat_in_range and lon_in_range


def calculate_boundary_info(nw_corner, se_corner):
    """Calculate boundary dimensions and area."""
    import math
    
    nw_lat, nw_lon = nw_corner
    se_lat, se_lon = se_corner
    
    # Calculate width and height in km
    # Width: distance along top edge
    width_km = haversine_distance((nw_lat, nw_lon), (nw_lat, se_lon))
    
    # Height: distance along left edge
    height_km = haversine_distance((nw_lat, nw_lon), (se_lat, nw_lon))
    
    # Approximate area
    area_km2 = width_km * height_km
    
    return {
        'width': width_km,
        'height': height_km,
        'area': area_km2
    }

    point_lat, point_lon = point
    nw_lat, nw_lon = nw_corner

    return closest, distance
```

```python
def find_furthest_location(depot, locations):
    """Find location furthest from depot."""
    max_distance = 0
    furthest = None
    
    for name, coord in locations.items():
        distance = haversine_distance(depot, coord)
        
        if distance > max_distance:
            max_distance = distance
            furthest = (name, coord)
    
    return furthest, max_distance


def calculate_round_trip(depot, destination):
    """Calculate round-trip distance."""
    one_way = haversine_distance(depot, destination)
    round_trip = one_way * 2
    
    return one_way, round_trip
```

```python
def calculate_gpa(grades):
    """Calculate GPA from list of grades (0-100 scale to 4.0 scale)."""
    if not grades:
        return 0.0
    
    average = sum(grades) / len(grades)
    
    # Convert to 4.0 scale
    if average >= 90:
        return 4.0
    elif average >= 80:
        return 3.0
    elif average >= 70:
        return 2.0
    elif average >= 60:
        return 1.0
    else:
        return 0.0

                    }

                student_data[name]['classes'].append({
                    'class': class_name,
                    'average': class_average,
                    'gpa': calculate_gpa(grades)
                })
                student_data[name]['all_grades'].extend(grades)
    
    # Calculate overall GPA
    for name, data in student_data.items():
        overall_average = sum(data['all_grades']) / len(data['all_grades'])
        data['overall_gpa'] = calculate_gpa(data['all_grades'])
```

### Question 17: Family Tree with Descendant Counter

**Problem:** Build a family tree structure using nested dictionaries. Each person has name, age, and children (list of person dictionaries). Write a function to count total descendants of a person.

**Solution Approach:**

- Recursive structure: person contains list of children (each also a person)
- Count descendants recursively: direct children + their descendants
- Base case: person with no children has 0 descendants

**Complete Solution:**

```python
def create_family_tree():
    """Create family tree with nested structure."""

def count_descendants(person):
    """Recursively count total descendants of a person."""
    if not person["children"]:
        return 0
    
    # Count direct children
    count = len(person["children"])
    
    # Add descendants of each child (recursive)
    for child in person["children"]:
        count += count_descendants(child)
    
    return count


def display_family_tree(person, level=0):
    """Display family tree with indentation."""
    indent = "  " * level
    
    descendant_count = count_descendants(person)
    print(f"{indent}{person['name']} (age {person['age']}) - {descendant_count} descendants")
    
    for child in person["children"]:
        display_family_tree(child, level + 1)


def find_person(family, name):
    """Find a person in the family tree by name."""
    if family["name"] == name:
        return family
    
    for child in family["children"]:
        result = find_person(child, name)
        if result:
            return result
    
    return None


def get_generation_counts(family):
    """Count people in each generation."""
    generations = {}
    
    def count_by_generation(person, generation=0):
        if generation not in generations:
            generations[generation] = []
        
        generations[generation].append(person["name"])
        
        for child in person["children"]:
            count_by_generation(child, generation + 1)
    
    count_by_generation(family)
    return generations


def calculate_family_statistics(family):
    """Calculate comprehensive family statistics."""
    all_people = []
    
    def collect_people(person):
        all_people.append(person)
        for child in person["children"]:
            collect_people(child)
    
    collect_people(family)
    
    ages = [p["age"] for p in all_people]
    
    stats = {
        "total_members": len(all_people),
        "average_age": sum(ages) / len(ages),
        "oldest": max(all_people, key=lambda p: p["age"]),
        "youngest": min(all_people, key=lambda p: p["age"]),
        "age_range": max(ages) - min(ages)
    }
    
    return stats
```

```python
        total_value += value
    
    return total_value
```

```python
def trace_team_path(bracket, team_name):
    """Trace a team's path through the tournament."""
    path = []
    
    def find_path(match):
        # Check if team played in this match
        if team_name in [match["team1"], match["team2"]]:
            result = "WON" if match["winner"] == team_name else "LOST"
            path.append({
                "round": match["round"],
                "opponent": match["team2"] if match["team1"] == team_name else match["team1"],
                "result": result
            })
            
            # If won, continue tracing
            if result == "WON" and "match1" not in match:
                return True
        
        # Recurse into sub-matches
        if "match1" in match:
            if find_path(match["match1"]):
                return True
        if "match2" in match:
            if find_path(match["match2"]):
                return True
        
        return False
    
    find_path(bracket)
    return path
```

```python
import json

class RecipeManager:
    """Complete recipe management system with JSON storage."""
    
    def __init__(self, filename="recipes.json"):
        self.filename = filename
        self.recipes = self.load_recipes()
    
    def load_recipes(self):
        """Load recipes from JSON file."""
        try:
            with open(self.filename, 'r') as f:
                return json.load(f)
        except FileNotFoundError:
            return {}
    
    def save_recipes(self):
        """Save recipes to JSON file."""
        with open(self.filename, 'w') as f:
            json.dump(self.recipes, f, indent=2)
    
    def add_recipe(self, name, ingredients, steps, cooking_time):
        """Add a new recipe."""
        self.recipes[name] = {
            "ingredients": ingredients,
            "steps": steps,
            "cooking_time": cooking_time
        }
        self.save_recipes()
        print(f"\nRecipe '{name}' added successfully!")
    
    def search_by_ingredient(self, ingredient):
        """Find all recipes containing specific ingredient."""
        results = []
        
        for recipe_name, recipe_data in self.recipes.items():
            for ing in recipe_data["ingredients"]:
                if ingredient.lower() in ing["name"].lower():
                    results.append(recipe_name)
                    break
        
        return results
    
    def scale_recipe(self, recipe_name, scale_factor):
        """Scale recipe quantities by factor."""
        if recipe_name not in self.recipes:
            return None
        
        recipe = self.recipes[recipe_name]
        scaled = recipe.copy()
        
        # Scale ingredients
        scaled["ingredients"] = []
        for ing in recipe["ingredients"]:
            scaled_ing = ing.copy()
            scaled_ing["quantity"] = ing["quantity"] * scale_factor
            scaled["ingredients"].append(scaled_ing)
        
        return scaled
```

## Question 22: Route Planning with Traveling Salesman Optimization

**Problem:** Build a route planning system that loads city coordinates from JSON, calculates distances between all cities, and finds the shortest path visiting all cities exactly once (traveling salesman problem - use simplified approach).

**Solution Approach:**

- Calculate distance matrix for all city pairs
- Use nearest neighbor heuristic for route optimization
- Compare with random routes to show improvement

**Complete Solution:**

```python
import json
import math
from itertools import permutations

def haversine_distance(coord1, coord2):
    """Calculate distance between coordinates in km."""
    lat1, lon1 = coord1
    lat2, lon2 = coord2
    
    R = 6371
    lat1, lon1, lat2, lon2 = map(math.radians, [lat1, lon1, lat2, lon2])
    
    dlat = lat2 - lat1
    dlon = lon2 - lon1
    a = math.sin(dlat/2)**2 + math.cos(lat1) * math.cos(lat2) * math.sin(dlon/2)**2
    c = 2 * math.asin(math.sqrt(a))
    
    return R * c

def build_distance_matrix(cities):
    """Build distance matrix for all city pairs."""
    city_names = list(cities.keys())
    distances = {}
    
    for i, city1 in enumerate(city_names):
        for j, city2 in enumerate(city_names):
            if i != j:
                coord1 = (cities[city1]["lat"], cities[city1]["lon"])
                coord2 = (cities[city2]["lat"], cities[city2]["lon"])
                dist = haversine_distance(coord1, coord2)
                distances[(city1, city2)] = dist
    
    return distances


def calculate_route_distance(route, distances):
    """Calculate total distance for a route."""
    total = 0
    
    for i in range(len(route) - 1):
        total += distances[(route[i], route[i + 1])]
    
    # Return to start
    total += distances[(route[-1], route[0])]
    
    return total


def nearest_neighbor_route(cities, distances, start_city):
    """Find route using nearest neighbor heuristic."""
    unvisited = set(cities.keys()) - {start_city}
    route = [start_city]
    current = start_city
    
    while unvisited:
        # Find nearest unvisited city
        nearest = min(unvisited, key=lambda city: distances[(current, city)])
        route.append(nearest)
        current = nearest
        unvisited.remove(nearest)
    
    return route


def find_optimal_route_bruteforce(cities, distances):
    """Find optimal route using brute force (small datasets only)."""
    city_names = list(cities.keys())
    start_city = city_names[0]
    remaining = city_names[1:]
    
    best_route = None
    best_distance = float('inf')
    
    # Try all permutations of remaining cities
    for perm in permutations(remaining):
        route = [start_city] + list(perm)
        dist = calculate_route_distance(route, distances)
        
        if dist < best_distance:
            best_distance = dist
            best_route = route
    
    return best_route, best_distance

def compare_routes(cities, distances):
    """Compare different routing strategies."""
    print(f"\n{'='*70}")
    print("ROUTE COMPARISON")
    print('='*70)
    
    city_names = list(cities.keys())
    
    # Strategy 1: Alphabetical order
    alphabetical_route = sorted(city_names)
    alpha_dist = calculate_route_distance(alphabetical_route, distances)
    
    # Strategy 2: Nearest neighbor
    nn_route = nearest_neighbor_route(cities, distances, city_names[0])
    nn_dist = calculate_route_distance(nn_route, distances)
    
    # Strategy 3: Brute force (if small enough)
    if len(cities) <= 7:
        optimal_route, optimal_dist = find_optimal_route_bruteforce(cities, distances)
    else:
        optimal_route = None
        optimal_dist = None
    
    print(f"\n{'Strategy':<30} {'Distance (km)':<15} {'Improvement':<15}")

    if optimal_route:

    return {

        'nearest_neighbor': (nn_route, nn_dist),
```

```python
def parse_time_slot(time_str):
    """Parse time slot string into structured format."""
    # Format: "MWF 09:00-10:00" or "TTh 14:00-15:30"
    parts = time_str.split()
    days = parts[0]
    time_range = parts[1]
    start_time, end_time = time_range.split('-')
    
    # Convert days to list
    day_list = []
    if 'M' in days:
        day_list.append('Monday')
    if 'Th' in days:
        day_list.append('Thursday')
    elif 'T' in days:
        day_list.append('Tuesday')
    if 'W' in days:
        day_list.append('Wednesday')
    if 'F' in days:
        day_list.append('Friday')
    
    return {
        'days': day_list,
        'start': start_time,
        'end': end_time
    }


def check_time_conflict(time1, time2):
    """Check if two time slots conflict."""
    slot1 = parse_time_slot(time1)
    slot2 = parse_time_slot(time2)
    
    # Check if any days overlap
    common_days = set(slot1['days']) & set(slot2['days'])
    if not common_days:
        return False
    
    # Check if times overlap
    start1 = datetime.strptime(slot1['start'], '%H:%M')
    end1 = datetime.strptime(slot1['end'], '%H:%M')
    start2 = datetime.strptime(slot2['start'], '%H:%M')
    end2 = datetime.strptime(slot2['end'], '%H:%M')
    
    # Times conflict if they overlap
    return not (end1 <= start2 or end2 <= start1)


def find_scheduling_conflicts(schedule, term):
    """Find all scheduling conflicts in a term."""
    if term not in schedule:
        return []
    
    conflicts = []
    sections = []
    
    # Collect all sections with course info
    for course_name, course_sections in schedule[term].items():

def calculate_room_utilization(schedule, term):
    """Calculate utilization percentage for each room."""
    if term not in schedule:
        return {}
    
    room_usage = {}
    
    for course_name, sections in schedule[term].items():
        for section in sections:
            room = section['room']

def enroll_student(schedule, term, course, section_id, student_name):
    """Enroll a student in a course section."""
    if term not in schedule or course not in schedule[term]:
        return False, "Course not found"
    
    for section in schedule[term][course]:
        if section['section'] == section_id:
            # Check capacity
            if len(section['enrolled']) >= section['capacity']:
                return False, "Section is full"
            
            # Check if already enrolled
            if student_name in section['enrolled']:
                return False, "Student already enrolled"
            
            # Enroll student
            section['enrolled'].append(student_name)
            return True, "Enrollment successful"
    
    return False, "Section not found"


def check_student_conflicts(schedule, term, student_name):
    """Check if student has any scheduling conflicts."""
    student_schedule = []
    
    for course_name, sections in schedule[term].items():
        for section in sections:
            if student_name in section['enrolled']:
                student_schedule.append({
                    'course': course_name,
                    'section': section['section'],
                    'time': section['time']
                })
    
    # Check for conflicts
    conflicts = []
    for i in range(len(student_schedule)):
        for j in range(i + 1, len(student_schedule)):
            if check_time_conflict(student_schedule[i]['time'], student_schedule[j]['time']):
                conflicts.append({
                    'course1': f"{student_schedule[i]['course']} {student_schedule[i]['section']}",
                    'course2': f"{student_schedule[j]['course']} {student_schedule[j]['section']}",
                    'time1': student_schedule[i]['time'],
                    'time2': student_schedule[j]['time']
                })
    
    return student_schedule, conflicts
```

**Solution Approach:**

```python
def generate_weather_data():
    """Generate sample historical weather data."""

    for day in range(180):
        current_date = start_date + timedelta(days=day)
        date_str = current_date.strftime('%Y-%m-%d')
        
        # Simulate seasonal temperature variation
        month = current_date.month
        base_temp = 20 + 10 * (month - 3.5) / 3.5  # Varies by season
        
        weather_data[date_str] = {
            'temperature': round(base_temp + (day % 10 - 5), 1),
            'humidity': round(60 + (day % 20), 1),

            'wind_speed': round(10 + (day % 15), 1),

        }

def calculate_monthly_averages(weather_data):
    """Calculate average values for each month."""
    monthly_data = defaultdict(lambda: {
        'temperature': [],
        'humidity': [],
        'precipitation': [],
        'wind_speed': [],
        'pressure': []
    })
    
    # Group data by month
    for date_str, measurements in weather_data.items():
        date = datetime.strptime(date_str, '%Y-%m-%d')
        month_key = date.strftime('%Y-%m')
        
        for metric, value in measurements.items():
            monthly_data[month_key][metric].append(value)
    
    # Calculate averages
    monthly_averages = {}
    for month, metrics in monthly_data.items():
        monthly_averages[month] = {}
        for metric, values in metrics.items():
            monthly_averages[month][metric] = sum(values) / len(values)
    
    return monthly_averages


def find_extreme_values(weather_data):
    """Find extreme values across all measurements."""
    extremes = {
        'temperature': {'max': float('-inf'), 'min': float('inf'), 'max_date': None, 'min_date': None},
        'humidity': {'max': float('-inf'), 'min': float('inf'), 'max_date': None, 'min_date': None},
        'precipitation': {'max': float('-inf'), 'min': float('inf'), 'max_date': None, 'min_date': None},
        'wind_speed': {'max': float('-inf'), 'min': float('inf'), 'max_date': None, 'min_date': None},
        'pressure': {'max': float('-inf'), 'min': float('inf'), 'max_date': None, 'min_date': None}
    }
    
    for date_str, measurements in weather_data.items():
        for metric, value in measurements.items():
            if value > extremes[metric]['max']:
                extremes[metric]['max'] = value
                extremes[metric]['max_date'] = date_str
            if value < extremes[metric]['min']:
                extremes[metric]['min'] = value
                extremes[metric]['min_date'] = date_str
    
    return extremes


def calculate_moving_average(weather_data, metric, window=7):
    """Calculate moving average for trend analysis."""
    sorted_dates = sorted(weather_data.keys())
    moving_avg = {}
    
    for i in range(len(sorted_dates)):
        if i < window - 1:
            continue
        
        window_dates = sorted_dates[i - window + 1:i + 1]
        window_values = [weather_data[date][metric] for date in window_dates]
        avg = sum(window_values) / len(window_values)
        moving_avg[sorted_dates[i]] = round(avg, 2)
    
    return moving_avg


def identify_trends(weather_data, metric):
    """Identify upward or downward trends."""
    moving_avg = calculate_moving_average(weather_data, metric, window=14)
    
    if not moving_avg:
        return None
    
    dates = sorted(moving_avg.keys())
    first_value = moving_avg[dates[0]]
    last_value = moving_avg[dates[-1]]
    
    change = last_value - first_value
    percent_change = (change / first_value) * 100 if first_value != 0 else 0
    
    if abs(percent_change) < 2:
        trend = "Stable"
    elif percent_change > 0:
        trend = "Increasing"
    else:
        trend = "Decreasing"
    
    return {
        'trend': trend,
        'change': round(change, 2),
        'percent_change': round(percent_change, 2),
        'start_value': round(first_value, 2),
        'end_value': round(last_value, 2)
    }

    """Get comprehensive weather summary."""
    total_days = len(weather_data)

    all_temps = [d['temperature'] for d in weather_data.values()]
    all_precip = [d['precipitation'] for d in weather_data.values()]
    
    avg_temp = sum(all_temps) / len(all_temps)
    total_precip = sum(all_precip)
    rainy_days = sum(1 for p in all_precip if p > 0)
    
    return {
        'total_days': total_days,
        'avg_temperature': round(avg_temp, 1),
        'total_precipitation': round(total_precip, 1),
        'rainy_days': rainy_days,
        'rainy_day_percent': round((rainy_days / total_days) * 100, 1)
    }


def find_weather_patterns(weather_data):
    """Find interesting weather patterns."""
    patterns = []
    sorted_dates = sorted(weather_data.keys())
    
    # Find consecutive hot days (temp > 25)
    hot_streak = 0
    max_hot_streak = 0
    
    for date in sorted_dates:
        if weather_data[date]['temperature'] > 25:
            hot_streak += 1
            max_hot_streak = max(max_hot_streak, hot_streak)
        else:
            hot_streak = 0
    
    if max_hot_streak > 0:
        patterns.append(f"Longest hot spell: {max_hot_streak} consecutive days above 25°C")
    
    # Find rainy days
    rainy_days = sum(1 for d in weather_data.values() if d['precipitation'] > 2)
    patterns.append(f"Heavy rain days (>2mm): {rainy_days}")
    
    # Find calm days (low wind)
    calm_days = sum(1 for d in weather_data.values() if d['wind_speed'] < 15)
    patterns.append(f"Calm days (wind <15 km/h): {calm_days}")
    
    return patterns
```

```python
import json
from datetime import datetime, timedelta

class EcommerceSystem:
    """Complete e-commerce system with inventory and order management."""
    
    def __init__(self, data_file='ecommerce_data.json'):
        self.data_file = data_file
        self.data = self.load_data()
    
    def load_data(self):

    def place_order(self, user_id, items):
        """
        Place a new order.
        items format: [{"product_id": "PROD001", "quantity": 2}, ...]
        """
        if user_id not in self.data["users"]:
            return None, "User not found"
        
        # Check inventory
        for item in items:
            product_id = item["product_id"]
            quantity = item["quantity"]
            
            if product_id not in self.data["products"]:
                return None, f"Product {product_id} not found"
            
            if self.data["products"][product_id]["inventory"] < quantity:
                return None, f"Insufficient inventory for {product_id}"
        
        # Calculate total
        order_items = []
        total_amount = 0
        
        for item in items:
            product = self.data["products"][item["product_id"]]
            item_total = product["price"] * item["quantity"]
            total_amount += item_total
            
            order_items.append({
                "product_id": item["product_id"],
                "product_name": product["name"],
                "quantity": item["quantity"],
                "price": product["price"],
                "subtotal": item_total
            })
            
            # Update inventory
            self.data["products"][item["product_id"]]["inventory"] -= item["quantity"]
        
        # Create order
        order_id = f"ORD{self.data['next_order_id']:05d}"
        self.data['next_order_id'] += 1

                "method": "Standard",
                "tracking": None
            },
            "date": datetime.now().strftime('%Y-%m-%d %H:%M:%S')
        }
        
        self.data["orders"][order_id] = order
        self.data["users"][user_id]["orders"].append(order_id)
        
        self.save_data()
        return order_id, "Order placed successfully"

        self.data["orders"][order_id]["status"] = new_status
        
        if tracking_number:

                    category = product["category"]
                    
                    if category not in category_revenue:

                            "revenue": 0,
                            "units_sold": 0,
                            "orders": 0
                        }
                    
                    category_revenue[category]["revenue"] += item["subtotal"]
                    category_revenue[category]["units_sold"] += item["quantity"]
                    category_revenue[category]["orders"] += 1
        
        return category_revenue

                    "category": product["category"],
                    "avg_rating": avg_rating,
                    "review_count": len(ratings),
                    "price": product["price"]
                })
        
        # Sort by rating
        product_ratings.sort(key=lambda x: (x["avg_rating"], x["review_count"]), reverse=True)
        
        return product_ratings[:n]

        review = {

            "date": datetime.now().strftime('%Y-%m-%d')
        }

        return True, "Review added successfully"

def display_order(order):
    """Display order details."""
    print(f"\n{'='*90}")
    print(f"ORDER: {order['order_id']}")
    print('='*90)
    
    print(f"\nDate: {order['date']}")
    print(f"Status: {order['status']}")
    print(f"User: {order['user_id']}")
    
    print("\nItems:")
    print("-" * 90)
    print(f"{'Product':<30} {'Quantity':<12} {'Price':<12} {'Subtotal':<12}")
    print("-" * 90)
    
    for item in order['items']:
        print(f"{item['product_name']:<30} {item['quantity']:<12} "
              f"${item['price']:<11.2f} ${item['subtotal']:<11.2f}")
    
    print("-" * 90)
    print(f"{'TOTAL':<54} ${order['total']:.2f}")
    
    print(f"\nShipping:")
    print(f"  Address: {order['shipping']['address']}")
    print(f"  Method: {order['shipping']['method']}")
    if order['shipping']['tracking']:
        print(f"  Tracking: {order['shipping']['tracking']}")
```
