def find_student(students, student_id):
    """Find a student by ID."""
    for student in students:
        if student["id"] == student_id:
            return student
    return None

### Filtering by Criteria

```python
def find_students_by_major(students, major):
    """Find all students with a specific major."""
    result = []
    for student in students:
        if student["major"] == major:
            result.append(student)
    return result

# Or using list comprehension (more elegant!)
cs_students = [s for s in students if s["major"] == "Computer Science"]

def find_students_by_age_range(students, min_age, max_age):
    """Find students within an age range."""
    return [s for s in students if min_age <= s["age"] <= max_age]
```

# Advanced Data Operations 📊

### Calculating Statistics

```python
def calculate_gpa(grades_dict):
    """Calculate GPA from grades dictionary."""
    if not grades_dict:
        return 0.0
    return sum(grades_dict.values()) / len(grades_dict)

def get_class_statistics(students):
    """Get comprehensive statistics for the class."""
    total_students = len(students)
    total_age = sum(student["age"] for student in students)
    average_age = total_age / total_students if total_students > 0 else 0

    # Major distribution
    major_counts = {}
    for student in students:
        major = student["major"]
        major_counts[major] = major_counts.get(major, 0) + 1

    # GPA statistics
    gpas = [calculate_gpa(student["grades"]) for student in students]
    average_gpa = sum(gpas) / len(gpas) if gpas else 0
    highest_gpa = max(gpas) if gpas else 0

    return {
        "total_students": total_students,
        "average_age": round(average_age, 1),
        "major_distribution": major_counts,
        "average_gpa": round(average_gpa, 2),
        "highest_gpa": round(highest_gpa, 2)
    }

# Usage
stats = get_class_statistics(students)
print(f"Class has {stats['total_students']} students")
print(f"Average age: {stats['average_age']}")
print(f"Average GPA: {stats['average_gpa']}")
```

```python
def enroll_student_in_course(course_enrollments, course_code, student_id):
    """Enroll a student in a course."""
    if course_code not in course_enrollments:
        return False, "Course not found"

    course = course_enrollments[course_code]
    enrolled = course["enrolled_students"]

    if student_id in enrolled:
        return False, "Student already enrolled"

    if len(enrolled) >= course["capacity"]:
        return False, "Course is full"

    enrolled.append(student_id)
    return True, "Student enrolled successfully"

def get_student_schedule(course_enrollments, student_id):
    """Get all courses and schedules for a student."""
    schedule = []

    for course_code, course_info in course_enrollments.items():
        if student_id in course_info["enrolled_students"]:
            schedule_info = {
                "course": course_code,
                "instructor": course_info["instructor"],
                "days": course_info["schedule"]["days"],
                "time": course_info["schedule"]["time"],
                "room": course_info["schedule"]["room"]
            }
            schedule.append(schedule_info)

    return schedule

# Usage
success, message = enroll_student_in_course(course_enrollments, "Math 101", "S020")
print(message)

student_schedule = get_student_schedule(course_enrollments, "S001")
for class_info in student_schedule:
    print(f"{class_info['course']}: {class_info['days']} at {class_info['time']}")
```

# Calculating Distances with GPS 📐

<v-clicks>

### The Haversine Formula

```python
import math

def haversine_distance(lat1, lon1, lat2, lon2):
    """
    Calculate the great circle distance between two points
    on Earth (specified in decimal degrees).

    Returns distance in kilometers.
    """
    # Convert decimal degrees to radians
    lat1, lon1, lat2, lon2 = map(math.radians, [lat1, lon1, lat2, lon2])

    # Haversine formula
    dlat = lat2 - lat1
    dlon = lon2 - lon1
    a = (math.sin(dlat/2)**2 +
         math.cos(lat1) * math.cos(lat2) * math.sin(dlon/2)**2)
    c = 2 * math.asin(math.sqrt(a))

    # Radius of Earth in kilometers
    earth_radius_km = 6371

    return earth_radius_km * c

# Calculate distances between cities
toronto = locations["toronto"]
vancouver = locations["vancouver"]

distance = haversine_distance(
    toronto["latitude"], toronto["longitude"],
    vancouver["latitude"], vancouver["longitude"]
)

print(f"Distance from Toronto to Vancouver: {distance:.1f} km")
```

# GPS Application: Location Finder 📍

<v-clicks>

```python
class LocationManager:
    """Manage GPS locations and calculate distances."""

    def __init__(self):
        self.locations = {}

    def add_location(self, name, latitude, longitude, **kwargs):
        """Add a new location with optional additional info."""
        self.locations[name] = {
            "latitude": latitude,
            "longitude": longitude,
            **kwargs  # Additional info like country, population, etc.
        }

    def get_distance(self, location1, location2):
        """Calculate distance between two locations."""
        if location1 not in self.locations or location2 not in self.locations:
            return None

        loc1 = self.locations[location1]
        loc2 = self.locations[location2]

        return haversine_distance(
            loc1["latitude"], loc1["longitude"],
            loc2["latitude"], loc2["longitude"]
        )

    def find_nearest_location(self, target_lat, target_lon, max_distance=None):
        """Find the nearest location to given coordinates."""
        nearest = None
        min_distance = float('inf')

        for name, location in self.locations.items():
            distance = haversine_distance(
                target_lat, target_lon,
                location["latitude"], location["longitude"]
            )

            if max_distance is None or distance <= max_distance:
                if distance < min_distance:
                    min_distance = distance
                    nearest = (name, location, distance)

        return nearest
```

# Complete Application: Travel Planner 🛣️

<v-clicks>

```python
import json
from datetime import datetime

class TravelPlanner:
    """Complete travel planning application."""

    def __init__(self):
        self.locations = {}
        self.trips = []

    def load_locations_from_json(self, filename):
        """Load locations from JSON file."""
        try:
            with open(filename, 'r') as file:
                data = json.load(file)
                self.locations = data
        except FileNotFoundError:
            print(f"File {filename} not found. Starting with empty locations.")

    def save_locations_to_json(self, filename):
        """Save locations to JSON file."""
        with open(filename, 'w') as file:
            json.dump(self.locations, file, indent=2)

    def plan_trip(self, trip_name, waypoints):
        """Plan a trip with multiple waypoints."""
        if not waypoints or len(waypoints) < 2:
            return None

        trip = {
            "name": trip_name,
            "waypoints": waypoints,
            "legs": [],
            "total_distance": 0,
            "created": datetime.now().isoformat()
        }

        # Calculate distances between consecutive waypoints
        for i in range(len(waypoints) - 1):
            from_location = waypoints[i]
            to_location = waypoints[i + 1]

            if from_location in self.locations and to_location in self.locations:
                distance = haversine_distance(
                    self.locations[from_location]["latitude"],
                    self.locations[from_location]["longitude"],
                    self.locations[to_location]["latitude"],
                    self.locations[to_location]["longitude"]
                )

                leg = {
                    "from": from_location,
                    "to": to_location,
                    "distance": round(distance, 1)
                }
                trip["legs"].append(leg)
                trip["total_distance"] += distance

        trip["total_distance"] = round(trip["total_distance"], 1)
        self.trips.append(trip)
        return trip
```

# Performance Considerations 🚀

<v-clicks>

### Choosing Efficient Data Structures

```python
# ❌ Inefficient: Linear search through list
def find_student_slow(students_list, student_id):
    for student in students_list:  # O(n) - gets slower with more students
        if student["id"] == student_id:
            return student
    return None

# ✅ Efficient: Dictionary lookup
def create_student_index(students_list):
    """Create a dictionary index for fast lookups."""
    return {student["id"]: student for student in students_list}

students_dict = create_student_index(students)

def find_student_fast(students_dict, student_id):
    return students_dict.get(student_id)  # O(1) - constant time!

# Benchmark the difference
import time

# For 10,000 students, dictionary lookup is 1000x faster!
```

### Memory vs Speed Trade-offs

```python
# Space-efficient but slower
def calculate_statistics_on_demand(students):
    """Calculate stats every time (saves memory, uses CPU)."""
    return {
        "count": len(students),
        "avg_age": sum(s["age"] for s in students) / len(students),
        "majors": set(s["major"] for s in students)
    }

# Memory-intensive but faster
def cache_statistics(students):
    """Pre-calculate and store stats (uses memory, saves CPU)."""
    cache = {
        "count": len(students),
        "avg_age": sum(s["age"] for s in students) / len(students),
        "majors": set(s["major"] for s in students),
        "by_major": {}
    }
    # Pre-group by major for fast filtering
    for student in students:
        major = student["major"]
        if major not in cache["by_major"]:
            cache["by_major"][major] = []
        cache["by_major"][major].append(student)

    return cache
```
