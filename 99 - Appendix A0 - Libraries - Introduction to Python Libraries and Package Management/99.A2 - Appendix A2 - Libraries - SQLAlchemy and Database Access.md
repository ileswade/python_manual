## Appendix A2 - Libraries - SQLAlchemy and Database Access

### Introduction

**SQLAlchemy** is Python's most popular SQL toolkit and Object-Relational Mapping (ORM) library. It allows you to interact with databases using Python code instead of writing raw SQL. This appendix covers both SQLAlchemy Core (SQL abstraction) and ORM (object-relational mapping) approaches, focusing on practical database operations.

## 2. Database Connection

### Connection Strings

SQLAlchemy uses connection strings (URLs) to connect to databases:

```python
from sqlalchemy import create_engine

# SQLite (file-based)
engine = create_engine('sqlite:///mydatabase.db')

# SQLite (in-memory)
engine = create_engine('sqlite:///:memory:')

# PostgreSQL
engine = create_engine('postgresql://user:password@localhost/dbname')

# MySQL
engine = create_engine('mysql+pymysql://user:password@localhost/dbname')

# SQL Server
engine = create_engine('mssql+pyodbc://user:password@localhost/dbname')
```

### Creating an Engine

```python
from sqlalchemy import create_engine

# Create engine (connection pool)
engine = create_engine('sqlite:///school.db', echo=True)

# echo=True shows all SQL statements (useful for learning)
# echo=False for production (default)
```

## 3. SQLAlchemy Core: Working with Tables

### Defining Tables with Core

```python
from sqlalchemy import create_engine, Table, Column, Integer, String, Float, MetaData

# Create engine
engine = create_engine('sqlite:///school.db')

# Metadata object holds table definitions
metadata = MetaData()

# Define students table
students = Table(
    'students',
    metadata,
    Column('id', Integer, primary_key=True),
    Column('name', String(100), nullable=False),
    Column('age', Integer),
    Column('grade', String(2)),
    Column('gpa', Float)
)

# Create tables in database
metadata.create_all(engine)
```

### Inserting Data

```python
from sqlalchemy import insert

# Single insert
stmt = insert(students).values(
    name='Alice Johnson',
    age=20,
    grade='A',
    gpa=3.8
)

with engine.connect() as conn:
    result = conn.execute(stmt)
    conn.commit()
    print(f"Inserted ID: {result.inserted_primary_key}")

# Multiple inserts
with engine.connect() as conn:
    conn.execute(
        insert(students),
        [
            {'name': 'Bob Smith', 'age': 21, 'grade': 'B', 'gpa': 3.5},
            {'name': 'Charlie Brown', 'age': 19, 'grade': 'A', 'gpa': 3.9},
            {'name': 'Diana Prince', 'age': 22, 'grade': 'A', 'gpa': 4.0}
        ]
    )
    conn.commit()
```

### Querying Data

```python
from sqlalchemy import select

# Select all
with engine.connect() as conn:
    result = conn.execute(select(students))
    for row in result:
        print(f"{row.id}: {row.name} - GPA: {row.gpa}")

# Select with WHERE clause
stmt = select(students).where(students.c.gpa >= 3.7)
with engine.connect() as conn:
    result = conn.execute(stmt)
    for row in result:
        print(f"{row.name}: {row.gpa}")

# Select specific columns
stmt = select(students.c.name, students.c.gpa)
with engine.connect() as conn:
    result = conn.execute(stmt)
    for row in result:
        print(f"{row.name}: {row.gpa}")
```

### Updating Data

```python
from sqlalchemy import update

# Update single record
stmt = update(students).where(
    students.c.name == 'Alice Johnson'
).values(gpa=3.9)

with engine.connect() as conn:
    conn.execute(stmt)
    conn.commit()

# Update multiple records
stmt = update(students).where(
    students.c.grade == 'A'
).values(gpa=4.0)

with engine.connect() as conn:
    result = conn.execute(stmt)
    conn.commit()
    print(f"Updated {result.rowcount} rows")
```

### Deleting Data

```python
from sqlalchemy import delete

# Delete with condition
stmt = delete(students).where(students.c.gpa < 2.0)

with engine.connect() as conn:
    result = conn.execute(stmt)
    conn.commit()
    print(f"Deleted {result.rowcount} rows")
```

## 4. SQLAlchemy ORM: Object-Relational Mapping

### Defining Models

```python
from sqlalchemy import create_engine, Column, Integer, String, Float
from sqlalchemy.orm import declarative_base, sessionmaker

# Create base class
Base = declarative_base()

# Define Student model
class Student(Base):
    __tablename__ = 'students'
    
    id = Column(Integer, primary_key=True)
    name = Column(String(100), nullable=False)
    age = Column(Integer)
    grade = Column(String(2))
    gpa = Column(Float)
    
    def __repr__(self):
        return f"<Student(id={self.id}, name='{self.name}', gpa={self.gpa})>"

# Create engine and tables
engine = create_engine('sqlite:///school_orm.db')
Base.metadata.create_all(engine)

# Create session factory
Session = sessionmaker(bind=engine)
```

### Creating and Adding Objects

```python
# Create session
session = Session()

# Create student objects
alice = Student(name='Alice Johnson', age=20, grade='A', gpa=3.8)
bob = Student(name='Bob Smith', age=21, grade='B', gpa=3.5)
charlie = Student(name='Charlie Brown', age=19, grade='A', gpa=3.9)

# Add to session
session.add(alice)
session.add(bob)
session.add(charlie)

# Or add multiple
session.add_all([alice, bob, charlie])

# Commit to database
session.commit()

print(f"Alice's ID: {alice.id}")  # ID assigned after commit
```

### Querying with ORM

```python
# Query all students
students = session.query(Student).all()
for student in students:
    print(student)

# Query with filter
high_performers = session.query(Student).filter(Student.gpa >= 3.7).all()
for student in high_performers:
    print(f"{student.name}: {student.gpa}")

# Get single student
alice = session.query(Student).filter_by(name='Alice Johnson').first()
print(alice)

# Get by primary key
student = session.query(Student).get(1)
print(student)

# Count
count = session.query(Student).count()
print(f"Total students: {count}")
```

### Updating with ORM

```python
# Update single object
alice = session.query(Student).filter_by(name='Alice Johnson').first()
alice.gpa = 3.9
session.commit()

# Update multiple
session.query(Student).filter(Student.grade == 'A').update({'gpa': 4.0})
session.commit()
```

### Deleting with ORM

```python
# Delete single object
student = session.query(Student).filter_by(name='Bob Smith').first()
session.delete(student)
session.commit()

# Delete multiple
session.query(Student).filter(Student.gpa < 2.0).delete()
session.commit()
```

### Closing Sessions

```python
# Always close sessions when done
session.close()

# Or use context manager
with Session() as session:
    students = session.query(Student).all()
    # Session automatically closes
```

## 5. Relationships Between Tables

### One-to-Many Relationship

```python
from sqlalchemy import ForeignKey
from sqlalchemy.orm import relationship

class Department(Base):
    __tablename__ = 'departments'
    
    id = Column(Integer, primary_key=True)
    name = Column(String(100), nullable=False)
    
    # Relationship to students
    students = relationship('Student', back_populates='department')
    
    def __repr__(self):
        return f"<Department(name='{self.name}')>"

class Student(Base):
    __tablename__ = 'students'
    
    id = Column(Integer, primary_key=True)
    name = Column(String(100), nullable=False)
    age = Column(Integer)
    grade = Column(String(2))
    gpa = Column(Float)
    department_id = Column(Integer, ForeignKey('departments.id'))
    
    # Relationship to department
    department = relationship('Department', back_populates='students')
    
    def __repr__(self):
        return f"<Student(name='{self.name}', gpa={self.gpa})>"

# Create tables
Base.metadata.create_all(engine)

# Use relationships
session = Session()

# Create department
cs_dept = Department(name='Computer Science')
session.add(cs_dept)
session.commit()

# Create students with department
alice = Student(name='Alice', age=20, gpa=3.8, department=cs_dept)
bob = Student(name='Bob', age=21, gpa=3.5, department=cs_dept)

session.add_all([alice, bob])
session.commit()

# Query through relationship
print(f"{cs_dept.name} has {len(cs_dept.students)} students")
for student in cs_dept.students:
    print(f"  - {student.name}")

# Access department from student
print(f"{alice.name} is in {alice.department.name}")
```

### Many-to-Many Relationship

```python
from sqlalchemy import Table

# Association table
student_courses = Table(
    'student_courses',
    Base.metadata,
    Column('student_id', Integer, ForeignKey('students.id')),
    Column('course_id', Integer, ForeignKey('courses.id'))
)

class Student(Base):
    __tablename__ = 'students'
    
    id = Column(Integer, primary_key=True)
    name = Column(String(100))
    
    # Many-to-many relationship
    courses = relationship('Course', secondary=student_courses, back_populates='students')

class Course(Base):
    __tablename__ = 'courses'
    
    id = Column(Integer, primary_key=True)
    name = Column(String(100))
    credits = Column(Integer)
    
    # Many-to-many relationship
    students = relationship('Student', secondary=student_courses, back_populates='courses')

# Create tables
Base.metadata.create_all(engine)

# Use many-to-many
session = Session()

# Create courses
python = Course(name='Python Programming', credits=3)
databases = Course(name='Database Systems', credits=4)
web_dev = Course(name='Web Development', credits=3)

# Create student
alice = Student(name='Alice')

# Enroll student in courses
alice.courses.append(python)
alice.courses.append(databases)

session.add_all([python, databases, web_dev, alice])
session.commit()

# Query relationships
print(f"{alice.name}'s courses:")
for course in alice.courses:
    print(f"  - {course.name} ({course.credits} credits)")

print(f"\nStudents in {python.name}:")
for student in python.students:
    print(f"  - {student.name}")
```

## 6. Advanced Queries

### Filtering and Ordering

```python
from sqlalchemy import and_, or_, desc

# Multiple conditions with and_
students = session.query(Student).filter(
    and_(Student.gpa >= 3.5, Student.age >= 20)
).all()

# Multiple conditions with or_
students = session.query(Student).filter(
    or_(Student.grade == 'A', Student.gpa >= 3.8)
).all()

# Order by
students = session.query(Student).order_by(Student.gpa.desc()).all()
students = session.query(Student).order_by(desc(Student.gpa)).all()

# Multiple order by
students = session.query(Student).order_by(
    Student.grade.asc(),
    Student.gpa.desc()
).all()

# Limit and offset
top_5 = session.query(Student).order_by(desc(Student.gpa)).limit(5).all()
next_5 = session.query(Student).order_by(desc(Student.gpa)).offset(5).limit(5).all()
```

### Aggregation

```python
from sqlalchemy import func

# Count
total = session.query(func.count(Student.id)).scalar()

# Average
avg_gpa = session.query(func.avg(Student.gpa)).scalar()

# Min and Max
min_gpa = session.query(func.min(Student.gpa)).scalar()
max_gpa = session.query(func.max(Student.gpa)).scalar()

# Sum
total_age = session.query(func.sum(Student.age)).scalar()

# Group by
grade_stats = session.query(
    Student.grade,
    func.count(Student.id).label('count'),
    func.avg(Student.gpa).label('avg_gpa')
).group_by(Student.grade).all()

for grade, count, avg_gpa in grade_stats:
    print(f"Grade {grade}: {count} students, avg GPA: {avg_gpa:.2f}")
```

### Joins

```python
# Inner join
results = session.query(Student, Department).join(Department).all()
for student, dept in results:
    print(f"{student.name} - {dept.name}")

# Left outer join
results = session.query(Student, Department).outerjoin(Department).all()

# Join with filter
cs_students = session.query(Student).join(Department).filter(
    Department.name == 'Computer Science'
).all()
```

## 7. Practical Examples

### Example 1: Student Management System

```python
from sqlalchemy import create_engine, Column, Integer, String, Float, Date
from sqlalchemy.orm import declarative_base, sessionmaker
from datetime import date

Base = declarative_base()
engine = create_engine('sqlite:///student_management.db')

class Student(Base):
    __tablename__ = 'students'
    
    id = Column(Integer, primary_key=True)
    name = Column(String(100), nullable=False)
    email = Column(String(100), unique=True)
    enrollment_date = Column(Date)
    gpa = Column(Float, default=0.0)
    
    def __repr__(self):
        return f"<Student(name='{self.name}', email='{self.email}')>"

Base.metadata.create_all(engine)
Session = sessionmaker(bind=engine)

def add_student(name, email):
    """Add new student."""
    session = Session()
    student = Student(
        name=name,
        email=email,
        enrollment_date=date.today()
    )
    session.add(student)
    session.commit()
    print(f"Added: {student}")
    session.close()

def get_all_students():
    """Get all students."""
    session = Session()
    students = session.query(Student).all()
    session.close()
    return students

def update_gpa(student_id, new_gpa):
    """Update student GPA."""
    session = Session()
    student = session.query(Student).get(student_id)
    if student:
        student.gpa = new_gpa
        session.commit()
        print(f"Updated {student.name}'s GPA to {new_gpa}")
    else:
        print("Student not found")
    session.close()

def delete_student(student_id):
    """Delete student."""
    session = Session()
    student = session.query(Student).get(student_id)
    if student:
        session.delete(student)
        session.commit()
        print(f"Deleted {student.name}")
    else:
        print("Student not found")
    session.close()

# Usage
add_student("Alice Johnson", "alice@email.com")
add_student("Bob Smith", "bob@email.com")
update_gpa(1, 3.8)

for student in get_all_students():
    print(student)
```

### Example 2: Blog System

```python
from sqlalchemy import Column, Integer, String, Text, DateTime, ForeignKey
from sqlalchemy.orm import relationship
from datetime import datetime

class User(Base):
    __tablename__ = 'users'
    
    id = Column(Integer, primary_key=True)
    username = Column(String(50), unique=True, nullable=False)
    email = Column(String(100), unique=True, nullable=False)
    created_at = Column(DateTime, default=datetime.utcnow)
    
    posts = relationship('Post', back_populates='author', cascade='all, delete-orphan')
    
    def __repr__(self):
        return f"<User(username='{self.username}')>"

class Post(Base):
    __tablename__ = 'posts'
    
    id = Column(Integer, primary_key=True)
    title = Column(String(200), nullable=False)
    content = Column(Text, nullable=False)
    created_at = Column(DateTime, default=datetime.utcnow)
    author_id = Column(Integer, ForeignKey('users.id'))
    
    author = relationship('User', back_populates='posts')
    
    def __repr__(self):
        return f"<Post(title='{self.title}')>"

Base.metadata.create_all(engine)

def create_user(username, email):
    """Create new user."""
    session = Session()
    user = User(username=username, email=email)
    session.add(user)
    session.commit()
    session.close()
    return user

def create_post(user_id, title, content):
    """Create new post."""
    session = Session()
    post = Post(title=title, content=content, author_id=user_id)
    session.add(post)
    session.commit()
    session.close()
    return post

def get_user_posts(user_id):
    """Get all posts by user."""
    session = Session()
    user = session.query(User).get(user_id)
    posts = user.posts if user else []
    session.close()
    return posts

# Usage
alice = create_user("alice", "alice@blog.com")
create_post(alice.id, "First Post", "This is my first blog post!")
create_post(alice.id, "Python Tips", "Here are some Python tips...")

posts = get_user_posts(alice.id)
for post in posts:
    print(f"{post.title} by {post.author.username}")
```

## 8. Best Practices

### Session Management

```python
# Bad - session might not close
def bad_example():
    session = Session()
    students = session.query(Student).all()
    return students  # Session not closed!

# Good - always close sessions
def good_example():
    session = Session()
    try:
        students = session.query(Student).all()
        return students
    finally:
        session.close()

# Better - use context manager
def best_example():
    with Session() as session:
        students = session.query(Student).all()
        return students
```

### Error Handling

```python
from sqlalchemy.exc import IntegrityError, SQLAlchemyError

def add_student_safe(name, email):
    """Add student with error handling."""
    session = Session()
    try:
        student = Student(name=name, email=email)
        session.add(student)
        session.commit()
        print(f"Added: {student}")
    except IntegrityError:
        session.rollback()
        print(f"Error: Email {email} already exists")
    except SQLAlchemyError as e:
        session.rollback()
        print(f"Database error: {e}")
    finally:
        session.close()
```

### Connection Pooling

```python
# Configure connection pool
engine = create_engine(
    'sqlite:///mydb.db',
    pool_size=10,           # Number of connections to maintain
    max_overflow=20,        # Additional connections if pool full
    pool_timeout=30,        # Seconds to wait for connection
    pool_recycle=3600       # Recycle connections after 1 hour
)
```

## Key Takeaways

✓ **SQLAlchemy Core** provides SQL abstraction layer

✓ **SQLAlchemy ORM** maps Python classes to database tables

✓ **Engine** manages database connections

✓ **Session** manages ORM transactions

✓ **Models** are Python classes that represent tables

✓ **Relationships** map foreign keys between tables

✓ **Always close sessions** to prevent resource leaks

✓ **Use context managers** for automatic session cleanup

✓ **Handle errors** with try-except blocks

✓ **ORM is higher-level** than Core but slightly slower
