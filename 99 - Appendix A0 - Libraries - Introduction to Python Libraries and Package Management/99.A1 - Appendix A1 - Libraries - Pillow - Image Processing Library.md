## 2. Opening and Saving Images

### Opening Images

```python
from PIL import Image

# Open an image
img = Image.open("photo.jpg")

# Display in default viewer
img.show()

# Get image information
print(f"Format: {img.format}")      # JPEG, PNG, etc.
print(f"Size: {img.size}")          # (width, height) tuple
print(f"Mode: {img.mode}")          # RGB, RGBA, L (grayscale), etc.
print(f"Width: {img.width}")
print(f"Height: {img.height}")
```

### Image Modes

Common modes:
- `'RGB'` - 3 channels (Red, Green, Blue)
- `'RGBA'` - 4 channels (RGB + Alpha/transparency)
- `'L'` - Grayscale (1 channel)
- `'1'` - Binary (black and white)
- `'CMYK'` - Cyan, Magenta, Yellow, Black

### Saving Images

```python
# Save with same format
img.save("output.jpg")

# Save with different format
img.save("output.png")

# Save with quality setting (JPEG only)
img.save("output.jpg", quality=95)  # 1-100

# Save with optimization
img.save("output.jpg", optimize=True)
```

### Supported Formats

Pillow supports many formats:
- **JPEG** (.jpg, .jpeg) - Lossy compression
- **PNG** (.png) - Lossless compression, supports transparency
- **GIF** (.gif) - Supports animation
- **BMP** (.bmp) - Uncompressed
- **TIFF** (.tiff) - Professional photography
- **WebP** (.webp) - Modern web format
- **ICO** (.ico) - Icons

#### Calculate Proportional Size

```python
def resize_proportional(image, max_width=None, max_height=None):
    """Resize maintaining aspect ratio."""
    width, height = image.size
    
    if max_width and max_height:
        # Fit within both dimensions
        ratio = min(max_width/width, max_height/height)
    elif max_width:
        ratio = max_width/width
    elif max_height:
        ratio = max_height/height
    else:
        return image
    
    new_size = (int(width * ratio), int(height * ratio))
    return image.resize(new_size)

img = Image.open("photo.jpg")
resized = resize_proportional(img, max_width=1920)
resized.save("photo_hd.jpg")
```

### Cropping Images

```python
# Crop format: (left, top, right, bottom)
img = Image.open("photo.jpg")

# Crop specific area
cropped = img.crop((100, 100, 400, 400))
cropped.save("photo_cropped.jpg")

# Crop center square
def crop_center_square(image):
    width, height = image.size
    min_dim = min(width, height)
    
    left = (width - min_dim) // 2
    top = (height - min_dim) // 2
    right = left + min_dim
    bottom = top + min_dim
    
    return image.crop((left, top, right, bottom))

square = crop_center_square(img)
square.save("photo_square.jpg")
```

### Watermark

```python
def add_watermark(image_path, watermark_text, output_path):
    """Add semi-transparent watermark to image."""
    img = Image.open(image_path).convert('RGBA')
    
    # Create transparent overlay
    txt_layer = Image.new('RGBA', img.size, (255, 255, 255, 0))
    draw = ImageDraw.Draw(txt_layer)
    
    # Try to load font
    try:
        font = ImageFont.truetype("arial.ttf", 48)
    except:
        font = ImageFont.load_default()
    
    # Calculate position (bottom right)
    bbox = draw.textbbox((0, 0), watermark_text, font=font)
    text_width = bbox[2] - bbox[0]
    text_height = bbox[3] - bbox[1]
    
    x = img.width - text_width - 20
    y = img.height - text_height - 20
    
    # Draw semi-transparent text
    draw.text((x, y), watermark_text, font=font, fill=(255, 255, 255, 128))
    
    # Composite images
    watermarked = Image.alpha_composite(img, txt_layer)
    watermarked.convert('RGB').save(output_path)

add_watermark("photo.jpg", "© 2024 My Photos", "photo_watermarked.jpg")
```

### Gradient Images

```python
from PIL import Image

def create_gradient(width, height, start_color, end_color):
    """Create vertical gradient."""
    gradient = Image.new('RGB', (width, height))
    draw = ImageDraw.Draw(gradient)
    
    for y in range(height):
        # Interpolate between colors
        ratio = y / height
        r = int(start_color[0] + (end_color[0] - start_color[0]) * ratio)
        g = int(start_color[1] + (end_color[1] - start_color[1]) * ratio)
        b = int(start_color[2] + (end_color[2] - start_color[2]) * ratio)
        
        draw.line([(0, y), (width, y)], fill=(r, g, b))
    
    return gradient

gradient = create_gradient(400, 300, (255, 0, 0), (0, 0, 255))
gradient.save("gradient.jpg")
```

## 9. Batch Processing

### Process Multiple Images

```python
import os
from PIL import Image

def batch_resize(input_folder, output_folder, max_size=(800, 800)):
    """Resize all images in folder."""
    if not os.path.exists(output_folder):
        os.makedirs(output_folder)
    
    for filename in os.listdir(input_folder):
        if filename.lower().endswith(('.png', '.jpg', '.jpeg', '.gif', '.bmp')):
            try:
                img_path = os.path.join(input_folder, filename)
                img = Image.open(img_path)
                
                img.thumbnail(max_size)
                
                output_path = os.path.join(output_folder, filename)
                img.save(output_path)
                print(f"Processed: {filename}")
            except Exception as e:
                print(f"Error processing {filename}: {e}")

batch_resize("original_photos", "resized_photos")
```

### Apply Filter to Multiple Images

```python
from PIL import ImageFilter

def batch_filter(input_folder, output_folder, filter_type='BLUR'):
    """Apply filter to all images."""
    if not os.path.exists(output_folder):
        os.makedirs(output_folder)
    
    filters = {
        'BLUR': ImageFilter.BLUR,
        'SHARPEN': ImageFilter.SHARPEN,
        'CONTOUR': ImageFilter.CONTOUR,
        'EMBOSS': ImageFilter.EMBOSS
    }
    
    filter_obj = filters.get(filter_type, ImageFilter.BLUR)
    
    for filename in os.listdir(input_folder):
        if filename.lower().endswith(('.png', '.jpg', '.jpeg')):
            img_path = os.path.join(input_folder, filename)
            img = Image.open(img_path)
            
            filtered = img.filter(filter_obj)
            
            output_path = os.path.join(output_folder, f"filtered_{filename}")
            filtered.save(output_path)
            print(f"Filtered: {filename}")

batch_filter("photos", "filtered_photos", filter_type='SHARPEN')
```

## 10. Advanced Techniques

### Instagram-Style Filters

```python
from PIL import Image, ImageEnhance, ImageFilter

def vintage_filter(image_path, output_path):
    """Apply vintage/retro filter."""
    img = Image.open(image_path)
    
    # Reduce saturation
    enhancer = ImageEnhance.Color(img)
    img = enhancer.enhance(0.5)
    
    # Increase contrast
    enhancer = ImageEnhance.Contrast(img)
    img = enhancer.enhance(1.3)
    
    # Slightly darker
    enhancer = ImageEnhance.Brightness(img)
    img = enhancer.enhance(0.9)
    
    # Slight blur
    img = img.filter(ImageFilter.GaussianBlur(radius=0.5))
    
    # Sepia tone
    grayscale = img.convert('L')
    sepia = ImageOps.colorize(grayscale, '#704214', '#C0A080')
    
    sepia.save(output_path)

def cold_filter(image_path, output_path):
    """Apply cold/blue filter."""
    img = Image.open(image_path)
    
    # Increase contrast
    enhancer = ImageEnhance.Contrast(img)
    img = enhancer.enhance(1.2)
    
    # Blue tint
    grayscale = img.convert('L')
    tinted = ImageOps.colorize(grayscale, '#1a1a2e', '#16213e', '#0f3460')
    
    # Blend with original
    blended = Image.blend(img.convert('RGB'), tinted, alpha=0.3)
    blended.save(output_path)

vintage_filter("photo.jpg", "photo_vintage.jpg")
cold_filter("photo.jpg", "photo_cold.jpg")
```

### Image Gallery Generator

```python
def create_gallery(image_folder, output_file, images_per_row=3, thumb_size=(200, 200)):
    """Create image gallery grid."""
    image_files = [f for f in os.listdir(image_folder)
                   if f.lower().endswith(('.png', '.jpg', '.jpeg'))]
    
    if not image_files:
        print("No images found!")
        return
    
    thumbnails = []
    for filename in image_files:
        img = Image.open(os.path.join(image_folder, filename))
        img.thumbnail(thumb_size)
        thumbnails.append(img)
    
    num_images = len(thumbnails)
    num_rows = (num_images + images_per_row - 1) // images_per_row
    
    gallery_width = thumb_size[0] * images_per_row
    gallery_height = thumb_size[1] * num_rows
    
    gallery = Image.new('RGB', (gallery_width, gallery_height), 'white')
    
    for idx, thumb in enumerate(thumbnails):
        row = idx // images_per_row
        col = idx % images_per_row
        x = col * thumb_size[0]
        y = row * thumb_size[1]
        gallery.paste(thumb, (x, y))
    
    gallery.save(output_file)
    print(f"Gallery saved to {output_file}")

create_gallery("my_photos", "gallery.jpg", images_per_row=4)
```
