# Understanding 'self' in Depth 🪞

Let's demystify **self** - the key to understanding how methods work:

<v-clicks>

### What Really Happens When You Call a Method

```python
class Counter:
    def __init__(self, start=0):
        self.count = start

    def increment(self):
        self.count += 1

    def get_count(self):
        return self.count

# Create object
my_counter = Counter(5)

# These two calls are equivalent:
my_counter.increment()              # What you write
Counter.increment(my_counter)       # What Python actually does!

# Python automatically passes the object as 'self'
```

**self** is just a reference to the object you're working with - nothing magical!

</v-clicks>

# 'self' in Action: Multiple Objects 👥

<v-clicks>

```python
class Dog:
    def __init__(self, name, breed):
        self.name = name
        self.breed = breed
        self.tricks = []

    def learn_trick(self, trick):
        self.tricks.append(trick)
        print(f"{self.name} learned {trick}!")

    def show_tricks(self):
        if self.tricks:
            print(f"{self.name} knows: {', '.join(self.tricks)}")
        else:
            print(f"{self.name} doesn't know any tricks yet")

# Create multiple dogs
buddy = Dog("Buddy", "Golden Retriever")
max_dog = Dog("Max", "German Shepherd")

# Each method call works with its specific object
buddy.learn_trick("sit")      # self = buddy in this call
max_dog.learn_trick("roll")   # self = max_dog in this call

buddy.show_tricks()    # Buddy knows: sit
max_dog.show_tricks()  # Max knows: roll
```

Each method call automatically receives the correct object as **self**!

</v-clicks>

# Methods That Work with Other Objects 🤝

<v-clicks>

Methods can interact with other objects of the same class:

```python
class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.balance = balance

    def transfer_to(self, other_account, amount):
        """Transfer money to another account."""
        if amount > 0 and amount <= self.balance:
            self.balance -= amount
            other_account.balance += amount
            print(f"Transferred ${amount} from {self.owner} to {other_account.owner}")
            return True
        else:
            print("Transfer failed: insufficient funds or invalid amount")
            return False

    def get_balance(self):
        return self.balance

# Create accounts
alice_account = BankAccount("Alice", 1000)
bob_account = BankAccount("Bob", 500)

# Transfer between accounts
alice_account.transfer_to(bob_account, 200)

print(f"Alice's balance: ${alice_account.get_balance()}")  # $800
print(f"Bob's balance: ${bob_account.get_balance()}")      # $700
```

**self** refers to the account doing the transfer, **other_account** is the recipient!

# Special Methods: Making Objects User-Friendly ✨

**Special methods** (also called "dunder methods" for double underscore) customize how objects behave:

<v-clicks>

### The Problem: Ugly Object Display

```python
class Student:
    def __init__(self, name, student_id, major):
        self.name = name
        self.student_id = student_id
        self.major = major

alice = Student("Alice Johnson", "S001", "Computer Science")
print(alice)  # <__main__.Student object at 0x...> - Not helpful!
```

### The Solution: __str__ Method

```python
class Student:
    def __init__(self, name, student_id, major):
        self.name = name
        self.student_id = student_id
        self.major = major

    def __str__(self):
        """Return a user-friendly string representation."""
        return f"{self.name} (ID: {self.student_id}) - Major: {self.major}"

alice = Student("Alice Johnson", "S001", "Computer Science")
print(alice)  # Alice Johnson (ID: S001) - Major: Computer Science
```

</v-clicks>

# __str__ vs __repr__: Two Types of String Representation 📝

<v-clicks>

### __str__: For End Users (Readable)

```python
class Book:
    def __init__(self, title, author, year):
        self.title = title
        self.author = author
        self.year = year

    def __str__(self):
        """User-friendly representation."""
        return f"'{self.title}' by {self.author} ({self.year})"

book = Book("Python Programming", "John Smith", 2023)
print(book)  # 'Python Programming' by John Smith (2023)
```

### __repr__: For Developers (Unambiguous)

```python
class Book:
    def __init__(self, title, author, year):
        self.title = title
        self.author = author
        self.year = year

    def __str__(self):
        """User-friendly representation."""
        return f"'{self.title}' by {self.author} ({self.year})"

    def __repr__(self):
        """Developer representation - should be unambiguous."""
        return f"Book('{self.title}', '{self.author}', {self.year})"

book = Book("Python Programming", "John Smith", 2023)
print(str(book))   # 'Python Programming' by John Smith (2023)
print(repr(book))  # Book('Python Programming', 'John Smith', 2023)
```

</v-clicks>

# When __str__ and __repr__ Are Used 🔍

<v-clicks>

```python
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __str__(self):
        return f"Point at ({self.x}, {self.y})"

    def __repr__(self):
        return f"Point({self.x}, {self.y})"

p1 = Point(3, 4)
p2 = Point(1, 2)

# __str__ is used by print()
print(p1)                    # Point at (3, 4)

# __repr__ is used in lists, debugging
points = [p1, p2]
print(points)                # [Point(3, 4), Point(1, 2)]

# Explicitly call each
print(str(p1))               # Point at (3, 4)
print(repr(p1))              # Point(3, 4)

# In interactive Python shell, __repr__ is used
>>> p1                       # Point(3, 4)
```

**Rule of thumb:**
- `__str__`: What you'd show to a user
- `__repr__`: What you'd show to a developer

</v-clicks>

# Comprehensive Class Example: Rectangle 📐

<v-clicks>

```python
class Rectangle:
    def __init__(self, width, height):
        """Initialize a rectangle with width and height."""
        self.width = width
        self.height = height

    def area(self):
        """Calculate the area of the rectangle."""
        return self.width * self.height

    def perimeter(self):
        """Calculate the perimeter of the rectangle."""
        return 2 * (self.width + self.height)

    def is_square(self):
        """Check if the rectangle is actually a square."""
        return self.width == self.height

    def scale(self, factor):
        """Scale the rectangle by a given factor."""
        self.width *= factor
        self.height *= factor

    def fits_inside(self, other_rectangle):
        """Check if this rectangle can fit inside another."""
        return (self.width <= other_rectangle.width and
                self.height <= other_rectangle.height)

    def __str__(self):
        """User-friendly string representation."""
        shape_type = "Square" if self.is_square() else "Rectangle"
        return f"{shape_type}: {self.width} × {self.height}"

    def __repr__(self):
        """Developer string representation."""
        return f"Rectangle({self.width}, {self.height})"
```

</v-clicks>

# Using the Rectangle Class 🎯

<v-clicks>

```python
# Create rectangles
rect1 = Rectangle(5, 3)
rect2 = Rectangle(4, 4)  # This is a square
rect3 = Rectangle(10, 8)

# Use the methods
print(f"Rectangle 1: {rect1}")                    # Rectangle: 5 × 3
print(f"Rectangle 2: {rect2}")                    # Square: 4 × 4
print(f"Area of rect1: {rect1.area()}")           # Area of rect1: 15
print(f"Perimeter of rect2: {rect2.perimeter()}") # Perimeter of rect2: 16

# Check properties
print(f"Is rect1 a square? {rect1.is_square()}")  # False
print(f"Is rect2 a square? {rect2.is_square()}")  # True

# Scale rectangle
print(f"Before scaling: {rect1}")                 # Rectangle: 5 × 3
rect1.scale(2)
print(f"After scaling by 2: {rect1}")             # Rectangle: 10 × 6

# Check if one fits inside another
print(f"Can rect2 fit inside rect3? {rect2.fits_inside(rect3)}")  # True
print(f"Can rect1 fit inside rect2? {rect1.fits_inside(rect2)}")  # False

# List of rectangles (shows __repr__)
rectangles = [rect1, rect2, rect3]
print(f"All rectangles: {rectangles}")
# All rectangles: [Rectangle(10, 6), Rectangle(4, 4), Rectangle(10, 8)]
```

# Method Categories: Different Types of Behavior 📊

<v-clicks>

### 1. Accessor Methods (Getters)
```python
class Circle:
    def __init__(self, radius):
        self.radius = radius

    def get_area(self):
        """Return calculated value without changing object."""
        return 3.14159 * self.radius ** 2

    def get_circumference(self):
        """Return calculated value without changing object."""
        return 2 * 3.14159 * self.radius
```

### 2. Mutator Methods (Setters)
```python
class Circle:
    def set_radius(self, new_radius):
        """Change the object's state."""
        if new_radius > 0:
            self.radius = new_radius
        else:
            print("Radius must be positive")

    def scale_by(self, factor):
        """Modify the object in place."""
        if factor > 0:
            self.radius *= factor
```

</v-clicks>

# More Method Categories 🔧

<v-clicks>

### 3. Query Methods (Boolean Returns)
```python
class Circle:
    def is_larger_than(self, other_circle):
        """Compare with another circle."""
        return self.radius > other_circle.radius

    def is_point_inside(self, x, y, center_x=0, center_y=0):
        """Check if a point is inside the circle."""
        distance_squared = (x - center_x)**2 + (y - center_y)**2
        return distance_squared <= self.radius**2
```

### 4. Factory Methods (Create Related Objects)
```python
class Circle:
    def create_inscribed_square(self):
        """Create a square that fits inside this circle."""
        from math import sqrt
        side_length = self.radius * 2 / sqrt(2)
        return Square(side_length)

    def create_concentric_circle(self, radius_ratio):
        """Create another circle with center at same point."""
        new_radius = self.radius * radius_ratio
        return Circle(new_radius)
```

# Real-World Example: Shopping Cart System 🛒

<v-clicks>

```python
class Product:
    def __init__(self, name, price, sku):
        self.name = name
        self.price = price
        self.sku = sku  # Stock Keeping Unit

    def __str__(self):
        return f"{self.name} - ${self.price:.2f}"

    def __repr__(self):
        return f"Product('{self.name}', {self.price}, '{self.sku}')"

class CartItem:
    def __init__(self, product, quantity=1):
        self.product = product
        self.quantity = quantity

    def get_total_price(self):
        return self.product.price * self.quantity

    def add_quantity(self, amount):
        if amount > 0:
            self.quantity += amount

    def remove_quantity(self, amount):
        if 0 < amount <= self.quantity:
            self.quantity -= amount
            return True
        return False

    def __str__(self):
        return f"{self.quantity}x {self.product.name} @ ${self.product.price:.2f} each"

    def __repr__(self):
        return f"CartItem({self.product!r}, {self.quantity})"
```

# Shopping Cart Implementation 🛍️

<v-clicks>

```python
class ShoppingCart:
    def __init__(self):
        self.items = []

    def add_product(self, product, quantity=1):
        """Add a product to the cart or update quantity if already exists."""
        # Check if product already in cart
        for item in self.items:
            if item.product.sku == product.sku:
                item.add_quantity(quantity)
                print(f"Updated {product.name} quantity to {item.quantity}")
                return

        # Add new item if not found
        new_item = CartItem(product, quantity)
        self.items.append(new_item)
        print(f"Added {quantity}x {product.name} to cart")

    def remove_product(self, sku, quantity=None):
        """Remove product from cart."""
        for i, item in enumerate(self.items):
            if item.product.sku == sku:
                if quantity is None or quantity >= item.quantity:
                    # Remove entire item
                    removed_item = self.items.pop(i)
                    print(f"Removed {removed_item.product.name} from cart")
                else:
                    # Reduce quantity
                    item.remove_quantity(quantity)
                    print(f"Reduced {item.product.name} quantity to {item.quantity}")
                return
        print("Product not found in cart")

    def get_total(self):
        """Calculate total cost of items in cart."""
        return sum(item.get_total_price() for item in self.items)

    def get_item_count(self):
        """Get total number of items (considering quantities)."""
        return sum(item.quantity for item in self.items)

    def clear_cart(self):
        """Remove all items from cart."""
        self.items.clear()
        print("Cart cleared")

    def __str__(self):
        if not self.items:
            return "Empty cart"

        cart_str = "Shopping Cart:\n"
        for item in self.items:
            cart_str += f"  {item} = ${item.get_total_price():.2f}\n"
        cart_str += f"Total: ${self.get_total():.2f} ({self.get_item_count()} items)"
        return cart_str

    def __repr__(self):
        return f"ShoppingCart({self.items!r})"
```

# Class Design Patterns 🎨

<v-clicks>

### 1. The Builder Pattern
```python
class Pizza:
    def __init__(self):
        self.size = "medium"
        self.crust = "thin"
        self.toppings = []

    def set_size(self, size):
        self.size = size
        return self  # Return self for method chaining!

    def set_crust(self, crust):
        self.crust = crust
        return self

    def add_topping(self, topping):
        self.toppings.append(topping)
        return self

    def __str__(self):
        return f"{self.size.title()} {self.crust} crust pizza with {', '.join(self.toppings)}"

# Method chaining makes building objects fluent
my_pizza = (Pizza()
            .set_size("large")
            .set_crust("thick")
            .add_topping("pepperoni")
            .add_topping("mushrooms")
            .add_topping("cheese"))

print(my_pizza)  # Large thick crust pizza with pepperoni, mushrooms, cheese
```

# More Design Patterns 🏗️

<v-clicks>

### 2. The Counter Pattern
```python
class WordCounter:
    def __init__(self):
        self.word_counts = {}
        self.total_words = 0

    def add_text(self, text):
        """Add text and count words."""
        words = text.lower().split()
        for word in words:
            self.word_counts[word] = self.word_counts.get(word, 0) + 1
            self.total_words += 1

    def get_count(self, word):
        """Get count for specific word."""
        return self.word_counts.get(word.lower(), 0)

    def get_most_common(self, n=5):
        """Get n most common words."""
        sorted_words = sorted(self.word_counts.items(),
                             key=lambda x: x[1], reverse=True)
        return sorted_words[:n]

    def __str__(self):
        return f"WordCounter: {self.total_words} total words, {len(self.word_counts)} unique"

# Usage
counter = WordCounter()
counter.add_text("the quick brown fox jumps over the lazy dog the fox is quick")

print(counter)  # WordCounter: 11 total words, 9 unique
print(f"'the' appears {counter.get_count('the')} times")  # 'the' appears 2 times
print(f"Most common: {counter.get_most_common(3)}")  # [('the', 2), ('quick', 2), ('fox', 2)]
```

# Best Practices for Intermediate Classes ✅

<v-clicks>

### 1. Always Implement __str__ and Often __repr__

```python
class Account:
    def __init__(self, account_number, balance):
        self.account_number = account_number
        self.balance = balance

    def __str__(self):
        return f"Account {self.account_number}: ${self.balance:.2f}"

    def __repr__(self):
        return f"Account('{self.account_number}', {self.balance})"
```

### 2. Make Methods Do One Thing Well

```python
# ❌ Method doing too much
def process_order(self):
    # Validate order
    # Calculate total
    # Apply discounts
    # Process payment
    # Send confirmation
    # Update inventory

# ✅ Split into focused methods

def process_payment(self): ...
def send_confirmation(self): ...
```
