### Question 13: Configuration File with Defaults
**Problem:** Write a program that reads a configuration file. If the file is missing or corrupted, use default settings.

**Solution Approach:**
- Define default configuration dictionary
- Try to read and parse config file
- Handle parsing errors gracefully
- Return defaults on any error
- Parse key=value format

**Complete Solution:**
```python
def load_config(filename):
    """Load configuration, use defaults if file missing or invalid."""
    
    default_config = {
        'host': 'localhost',
        'port': 8080,
        'debug': False,
        'timeout': 30
    }
    
    try:
        config = {}
        
        with open(filename, 'r') as file:
            for line_num, line in enumerate(file, start=1):
                line = line.strip()
                
                if not line or line.startswith('#'):
                    continue
                
                if '=' in line:
                    key, value = line.split('=', 1)
                    key = key.strip()
                    value = value.strip()
                    
                    # Convert value types
                    if value.lower() == 'true':
                        value = True
                    elif value.lower() == 'false':
                        value = False
                    elif value.isdigit():
                        value = int(value)
                    
                    config[key] = value
        
        print(f"✓ Configuration loaded from '{filename}'")
        return config
        
    except FileNotFoundError:
        print(f"Configuration file '{filename}' not found")
        print("Using default configuration")
        return default_config
        
    except Exception as e:
        print(f"Error reading configuration: {e}")
        print("Using default configuration")
        return default_config


# Use the function
config = load_config("app.config")

print("\n=== Current Configuration ===")
for key, value in config.items():
    print(f"{key}: {value}")
```

### Question 14: Backup and Restore File
**Problem:** Create a backup function that copies a file before modifying it. Restore the backup if modification fails.

**Solution Approach:**
- Create backup before modification
- Try to modify original file
- If modification fails, restore from backup
- Clean up backup on success
- Use .backup extension

**Complete Solution:**
```python
import shutil
import os

def safe_modify_file(filename, new_content):
    """Safely modify file with backup and rollback."""
    
    backup_file = filename + '.backup'
    
    # Step 1: Create backup
    if os.path.exists(filename):
        try:
            shutil.copy(filename, backup_file)
            print(f"✓ Backup created: {backup_file}")
        except IOError as e:
            print(f"Error creating backup: {e}")
            return False
    
    # Step 2: Try to modify
    try:
        with open(filename, 'w') as file:
            file.write(new_content)
        
        print(f"✓ File '{filename}' modified successfully")
        
        # Remove backup after success
        if os.path.exists(backup_file):
            os.remove(backup_file)
            print(f"✓ Backup removed")
        
        return True
        
    except IOError as e:
        print(f"✗ Error modifying file: {e}")
        
        # Step 3: Restore from backup
        if os.path.exists(backup_file):
            try:
                shutil.copy(backup_file, filename)
                print(f"✓ Restored from backup")
            except IOError:
                print(f"✗ Failed to restore backup")
        
        return False


# Test the function
original_content = "Important data\nDo not lose this\n"

with open("important.txt", 'w') as f:
    f.write(original_content)

print("=== Original File Created ===\n")

new_content = "Updated data\nNew information\n"
success = safe_modify_file("important.txt", new_content)

if success:
    print("\n✓ Operation successful")
else:
    print("\n✗ Operation failed, original preserved")
```

**Sample Output:**
```
=== Original File Created ===

✓ Backup created: important.txt.backup
✓ File 'important.txt' modified successfully
✓ Backup removed

✓ Operation successful
```

**Key Concepts:**
- Always backup before destructive operations
- `shutil.copy()` duplicates files
- Restore backup if modification fails

### Question 15: Merge Multiple Files
**Problem:** Build a file merger that combines multiple text files into one, handling missing files gracefully.

**Solution Approach:**
- Loop through list of input files
- Read each file that exists
- Skip missing files, continue with others
- Add separators between file contents
- Report successes and failures

**Complete Solution:**
```python
def merge_files(input_files, output_file):
    """
    Merge multiple files into one.
    Returns: (success_count, failed_files)
    """
    merged_content = []
    failed_files = []
    success_count = 0
    
    # Read each input file
    for filename in input_files:
        try:
            with open(filename, 'r') as file:
                content = file.read()
                
                # Add separator and source info
                merged_content.append(f"\n{'='*50}\n")
                merged_content.append(f"From: {filename}\n")
                merged_content.append(f"{'='*50}\n\n")
                merged_content.append(content)
                merged_content.append("\n")
                
                success_count += 1
                print(f"✓ Read '{filename}'")
                
        except FileNotFoundError:
            print(f"✗ File not found: '{filename}' (skipping)")
            failed_files.append(filename)
        except IOError as e:
            print(f"✗ Error reading '{filename}': {e} (skipping)")
            failed_files.append(filename)
    
    # Write merged content
    if merged_content:
        try:
            with open(output_file, 'w') as file:
                file.writelines(merged_content)
            
            print(f"\n✓ Merged {success_count} file(s) into '{output_file}'")
            return success_count, failed_files
            
        except IOError as e:
            print(f"\n✗ Error writing output file: {e}")
            return 0, input_files
    else:
        print("\n✗ No files could be read")
        return 0, input_files


# Test the function
files_to_merge = [
    "file1.txt",
    "file2.txt",
    "missing.txt",  # This will be skipped
    "file3.txt"
]

success, failed = merge_files(files_to_merge, "merged_output.txt")

print(f"\n=== Summary ===")
print(f"Successfully merged: {success}")
print(f"Failed: {len(failed)}")
if failed:
    print(f"Failed files: {', '.join(failed)}")
```

**Sample Output:**

```
✗ File not found: 'missing.txt' (skipping)
✓ Read 'file3.txt'

✓ Merged 3 file(s) into 'merged_output.txt'

=== Summary ===
Successfully merged: 3

```

**Key Concepts:**
- Continue processing despite errors

```python
def display_csv_table(filename):
    """Read CSV and display as formatted table."""
    
    try:
        with open(filename, 'r') as file:
            lines = file.readlines()
        
        if not lines:
            print("File is empty")
            return
        
        # Parse data
        rows = []
        for line in lines:

            # Add separator after header
            if i == 0:
                print("-" * total_width)
```

```python
                    log_counts['WARNING'] += 1
                elif 'ERROR:' in line:
                    log_counts['ERROR'] += 1
                    # Extract error message

                report.write("=" * 50 + "\n\n")

            report.write("\n" + "=" * 50 + "\n")
            report.write("SUMMARY\n")
            report.write("=" * 50 + "\n\n")

                report.write(f"⚠ {log_counts['ERROR']} errors detected!\n")

                report.write("✓ No errors found\n")
            
            if log_counts['WARNING'] > 0:
                report.write(f"⚠ {log_counts['WARNING']} warnings found\n")

        return True
        
    except IOError as e:
```

```python
def analyze_numbers(input_file, output_file):
    """Read numbers, calculate statistics, write results."""
    
    numbers = []
    errors = []
    
    # Read and parse numbers
    try:
        with open(input_file, 'r') as file:
            for line_num, line in enumerate(file, start=1):
                line = line.strip()
                
                if not line:
                    continue
                
                try:
                    number = float(line)
                    numbers.append(number)
                except ValueError:
                    errors.append(f"Line {line_num}: '{line}' is not a valid number")
        
        print(f"✓ Read {len(numbers)} valid number(s)")

            print(f"⚠ {len(errors)} line(s) skipped due to errors")

    average = total / count
    minimum = min(numbers)
    maximum = max(numbers)
    
    # Calculate median
    sorted_nums = sorted(numbers)

    if count % 2 == 0:
        median = (sorted_nums[mid-1] + sorted_nums[mid]) / 2
```

```python
def process_file(input_file, output_file):
    """Remove duplicates, sort, and save."""
    
    try:
        # Read all lines
        with open(input_file, 'r') as file:
            lines = file.readlines()
        
        original_count = len(lines)

        # Write results
        with open(output_file, 'w') as file:
            for line in sorted_lines:
                file.write(line + '\n')
        
        duplicates_removed = len(processed_lines) - unique_count
        
        print(f"✓ Removed {duplicates_removed} duplicate(s)")
        print(f"✓ Sorted {unique_count} unique line(s)")
        print(f"✓ Results saved to '{output_file}'")
        
        return True
        
    except FileNotFoundError:
        print(f"Error: File '{input_file}' not found")
        return False
```

### Question 24: Student Gradebook System
**Problem:** Create a gradebook system that saves student grades to JSON. Implement functions to add students, update grades, and calculate averages.

**Solution Approach:**
- Load/create gradebook from JSON
- Add student function
- Update grades function
- Calculate statistics
- Save changes to JSON
- Menu-driven interface

**Complete Solution:**
```python
import json
import os

def load_gradebook(filename):
    """Load gradebook from JSON."""
    if os.path.exists(filename):
        try:
            with open(filename, 'r') as file:
                return json.load(file)
        except (json.JSONDecodeError, IOError):
            return {}
    return {}


def save_gradebook(filename, gradebook):
    """Save gradebook to JSON."""
    try:
        with open(filename, 'w') as file:
            json.dump(gradebook, file, indent=4)
        return True
    except IOError:
        return False


def add_student(gradebook):
    """Add new student."""
    name = input("Student name: ").strip()
    
    if not name:
        print("Error: Name required")
        return
    
    if name in gradebook:
        print(f"Error: '{name}' already exists")
        return
    
    gradebook[name] = {
        "grades": [],
        "average": 0.0
    }
    
    print(f"✓ Added '{name}'")


def add_grade(gradebook):
    """Add grade for student."""
    name = input("Student name: ").strip()
    
    if name not in gradebook:
        print(f"Error: '{name}' not found")
        return
    
    try:
        grade = float(input("Grade (0-100): "))
        
        if grade < 0 or grade > 100:
            print("Error: Grade must be 0-100")
            return
        
        gradebook[name]["grades"].append(grade)
        
        # Recalculate average
        grades = gradebook[name]["grades"]
        gradebook[name]["average"] = sum(grades) / len(grades)
        
        print(f"✓ Grade added for '{name}'")
        
    except ValueError:
        print("Error: Invalid grade")


def view_gradebook(gradebook):
    """Display all students and grades."""
    if not gradebook:
        print("\nNo students in gradebook")
        return
    
    print(f"\n{'='*60}")
    print("GRADEBOOK")
    print(f"{'='*60}\n")
    
    for name, data in sorted(gradebook.items()):
        grades = data["grades"]
        avg = data["average"]
        
        print(f"{name}")
        print(f"  Grades: {grades}")
        print(f"  Average: {avg:.2f}")
        print()


def class_statistics(gradebook):
    """Calculate class statistics."""
    if not gradebook:
        print("\nNo data for statistics")
        return
    
    all_averages = [data["average"] for data in gradebook.values() if data["grades"]]
    
    if not all_averages:
        print("\nNo grades entered yet")
        return
    
    print(f"\n{'='*60}")
    print("CLASS STATISTICS")
    print(f"{'='*60}\n")
    print(f"Total students: {len(gradebook)}")
    print(f"Class average: {sum(all_averages)/len(all_averages):.2f}")
    print(f"Highest average: {max(all_averages):.2f}")
    print(f"Lowest average: {min(all_averages):.2f}")


def main():
    """Main program."""
    filename = "gradebook.json"
    gradebook = load_gradebook(filename)
    
    print(f"✓ Loaded gradebook with {len(gradebook)} student(s)\n")

        print("3. View gradebook")
        print("4. Class statistics")
        print("5. Save and exit")
        
        choice = input("\nChoice: ").strip()
        
        if choice == '1':
            add_student(gradebook)
        elif choice == '2':
            add_grade(gradebook)
        elif choice == '3':
            view_gradebook(gradebook)
        elif choice == '4':
            class_statistics(gradebook)
        elif choice == '5':
            if save_gradebook(filename, gradebook):
                print("✓ Gradebook saved")
                break
            else:
                print("Error saving gradebook")
        else:
            print("Invalid choice")


# Run program
main()
```

```python
import os
import shutil

def read_file():
    """Read and display file contents."""
    filename = input("Enter filename to read: ")
    
    try:
        with open(filename, 'r') as file:
            content = file.read()
            
        print(f"\n{'='*60}")
        print(f"Contents of '{filename}'")
        print(f"{'='*60}\n")
        print(content)
        print(f"\n{'='*60}")
        print(f"Total characters: {len(content)}")
        
    except FileNotFoundError:
        print(f"Error: File '{filename}' not found")
    except PermissionError:
        print(f"Error: No permission to read '{filename}'")
    except IOError as e:
        print(f"Error reading file: {e}")


def write_file():
    """Write content to file."""
    filename = input("Enter filename to write: ")
    
    if os.path.exists(filename):
        confirm = input(f"'{filename}' exists. Overwrite? (yes/no): ")
        if confirm.lower() != 'yes':
            print("Operation cancelled")
            return
    
    print("Enter content (type 'END' on a new line to finish):")
    lines = []
    
    while True:
        line = input()
        if line == 'END':
            break
        lines.append(line)
    
    try:
        with open(filename, 'w') as file:
            file.write('\n'.join(lines))
        
        print(f"✓ Wrote {len(lines)} line(s) to '{filename}'")
        
    except PermissionError:
        print(f"Error: No permission to write to '{filename}'")
    except IOError as e:
        print(f"Error writing file: {e}")


def append_file():
    """Append content to file."""
    filename = input("Enter filename to append to: ")
    
    print("Enter content to append (type 'END' on a new line to finish):")
    lines = []
    
    while True:
        line = input()
        if line == 'END':
            break
        lines.append(line)
    
    try:
        with open(filename, 'a') as file:
            file.write('\n'.join(lines) + '\n')
        
        print(f"✓ Appended {len(lines)} line(s) to '{filename}'")
        
    except PermissionError:
        print(f"Error: No permission to append to '{filename}'")
    except IOError as e:
        print(f"Error appending to file: {e}")


def copy_file():
    """Copy file to new location."""
    source = input("Enter source filename: ")
    destination = input("Enter destination filename: ")
    
    if not os.path.exists(source):
        print(f"Error: Source file '{source}' not found")
        return
    
    if os.path.exists(destination):
        confirm = input(f"'{destination}' exists. Overwrite? (yes/no): ")
        if confirm.lower() != 'yes':
            print("Operation cancelled")
            return
    
    try:
        shutil.copy(source, destination)
        
        size = os.path.getsize(destination)
        print(f"✓ Copied '{source}' to '{destination}'")
        print(f"  Size: {size} bytes")
        
    except PermissionError:
        print("Error: Permission denied")
    except IOError as e:
        print(f"Error copying file: {e}")


def delete_file():
    """Delete a file."""
    filename = input("Enter filename to delete: ")
    
    if not os.path.exists(filename):
        print(f"Error: File '{filename}' not found")
        return
    
    # Show file info before deletion
    try:
        size = os.path.getsize(filename)
        print(f"\nFile: {filename}")
        print(f"Size: {size} bytes")
        
        confirm = input("\nAre you sure you want to delete? (yes/no): ")
        
        if confirm.lower() == 'yes':
            os.remove(filename)
            print(f"✓ Deleted '{filename}'")
        else:
            print("Operation cancelled")
            
    except PermissionError:
        print(f"Error: No permission to delete '{filename}'")
    except IOError as e:
        print(f"Error: {e}")


def list_files():
    """List files in current directory."""
    try:
        files = os.listdir('.')
        
        print(f"\n{'='*60}")
        print("Files in current directory")
        print(f"{'='*60}\n")
        
        file_count = 0
        dir_count = 0
        
        for item in sorted(files):
            if os.path.isfile(item):

                print(f"  {item} ({size} bytes)")

            elif os.path.isdir(item):
                print(f"  [{item}]/")
                dir_count += 1
        
        print(f"\n{'='*60}")
        print(f"Total: {file_count} file(s), {dir_count} folder(s)")
        
    except IOError as e:
        print(f"Error listing directory: {e}")


def file_info():

    if not os.path.exists(filename):
        print(f"Error: '{filename}' not found")
        return
    
    try:

        modified = os.path.getmtime(filename)

        print(f"File Information: {filename}")
        print(f"{'='*60}\n")
        print(f"Size: {size} bytes")
        print(f"Modified: {mod_time.strftime('%Y-%m-%d %H:%M:%S')}")
        print(f"Path: {os.path.abspath(filename)}")
        
    except IOError as e:
        print(f"Error: {e}")


def main():

    print("="*60)
    print("COMPREHENSIVE FILE MANAGER")
    print("="*60)

        print("3. Append to file")
        print("4. Copy file")
        print("5. Delete file")
        print("6. List files")
        print("7. File info")
        print("8. Exit")
        
        choice = input("\nChoice: ").strip()
        print()
        
        if choice == '1':
            read_file()
        elif choice == '2':
            write_file()
        elif choice == '3':

        elif choice == '5':

            break
        else:
            print("Invalid choice")

main()
```

### Common Patterns

7. **Check file existence** - Use `os.path.exists()`
8. **Confirm destructive operations** - Ask before deleting/overwriting
